I wrote a lightweight subscriber to monitor color mode mismatches on client mounts:

export const watchContrastTokens = (theme: ThemeMode) => {
  const root = document.documentElement;
  const required = ['--on-surface', '--surface-container'];

  return required.every(key => {
    return getComputedStyle(root).getPropertyValue(key).length > 0;
  });
};
